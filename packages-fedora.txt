# Packages for Fedora:
#   sudo dnf install $(grep -v '^#' packages-fedora.txt)
#
# Enable these COPRs first:
#   sudo dnf copr enable mineiro/hyprland            # Hyprland, hypr tools, awww, cliphist, uwsm
#   sudo dnf copr enable errornointernet/quickshell
#   sudo dnf copr enable erikreider/SwayNotificationCenter
#   sudo dnf copr enable tofik/nwg-shell             # nwg-look
#   sudo dnf copr enable komapro/nerd-fonts          # 0xProto Nerd Font
#   sudo dnf copr enable lihaohong/yazi
#   sudo dnf copr enable peterwu/rendezvous          # Bibata cursors
#
# Every name below was checked against those repos on Fedora 44.
#
# Not packaged, install by hand:
#   matugen      cargo install matugen
#   oh-my-posh   curl -s https://ohmyposh.dev/install.sh | bash -s -- -d ~/.local/bin
#   nwg-displays python3 -m pip install --user git+https://github.com/nwg-piotr/nwg-displays
#   kora icons   git clone --depth 1 https://github.com/bikass/kora.git, copy kora/kora to ~/.local/share/icons
#   grimblast    from hyprwm/contrib (screenshot.sh uses it for some modes)
#   flatpaks     flatpak install flathub org.gnome.Calculator com.tomjwatson.Emote
